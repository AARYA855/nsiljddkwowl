        <div id="messages"></div>

        <input type="text" id="input" placeholder="Enter message">

        <button onclick="reply()">Send</button>

    </div>

    <script>

        function reply() {

            let msg = document.getElementById("input").value.toLowerCase();

            let response = "Sorry, I don't understand.";

            if(msg.includes("hi") || msg.includes("hello"))
                response = "Hello Customer!";

            else if(msg.includes("price"))
                response = "Price starts from Rs.500";

            else if(msg.includes("delivery"))
                response = "Delivery in 3 days";

            document.getElementById("messages").innerHTML +=
                "<p><b>You:</b> " + msg + "</p>" +
                "<p><b>Bot:</b> " + response + "</p>";

            document.getElementById("input").value = "";
        }

    </script>
