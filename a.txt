#include <iostream>
#include <queue>
#include <map>
#include <vector>
#include <cmath>

using namespace std;

struct Node {

    char vertex;
    int cost;

    bool operator<(const Node& other) const {
        return cost > other.cost;
    }
};

// Coordinates
map<char, pair<int,int>> pos = {

    {'A',{0,0}},
    {'B',{1,0}},
    {'C',{0,1}},
    {'D',{1,1}}
};

// Goal node
char goal = 'D';

// Manhattan heuristic
int h(char n) {

    return abs(pos[n].first - pos[goal].first) +
           abs(pos[n].second - pos[goal].second);
}

int main() {

    // Graph
    map<char, vector<pair<char,int>>> graph;

    graph['A'] = {{'B',1}, {'C',3}};
    graph['B'] = {{'D',2}};
    graph['C'] = {{'D',1}};

    priority_queue<Node> pq;

    map<char,bool> visited;

    pq.push({'A', h('A')});

    cout << "A* Traversal:\n";

    while(!pq.empty()) {

        Node current = pq.top();
        pq.pop();

        char v = current.vertex;

        if(visited[v])
            continue;

        visited[v] = true;

        cout << "\nNode: " << v;
        cout << "\nHeuristic: " << h(v) << endl;

        if(v == goal) {

            cout << "\nGoal Reached";
            break;
        }

        for(auto neighbor : graph[v]) {

            char next = neighbor.first;

            int totalCost =
                neighbor.second + h(next);

            cout << next
                 << " Cost = "
                 << totalCost << endl;

            pq.push({next, totalCost});
        }
    }

    return 0;
}
